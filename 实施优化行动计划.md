```java
public void handleSessionExpire(String clientId) {

    Set<MqttSubscription> subscriptions = subscriptionsDirectory.getSubscriptions(clientId);
    
    // 2. 遍历并删除每个订阅
```

```java
@Component
public class AclManager {
    
    private static final Logger logger = LoggerFactory.getLogger(AclManager.class);
    private Map<String, AclRule> rules = new ConcurrentHashMap<>();
    
    @Autowired
    private RedisTemplate<String, String> redisTemplate;
    
    @Value("${mqtt.acl.file:acl.conf}")
    private String aclFile;
    
    @Value("${mqtt.acl.reloadSeconds:30}")
    private long reloadSeconds;
    
    @PostConstruct
    public void init() {
        loadRulesFromFile();
        // 定期重新加载
        ScheduledExecutorService executor = Executors.newScheduledThreadPool(1);
        executor.scheduleAtFixedRate(this::loadRulesFromFile, 
            reloadSeconds, reloadSeconds, TimeUnit.SECONDS);
    }
    
    /**
     * 检查用户是否可以发布到指定主题
     * @param userId 用户ID
     * @param topic 主题路径
     * @return 是否允许
     */
    public boolean canPublish(String userId, String topic) {
        return checkPermission(userId, topic, "pub");
    }
    
    /**
     * 检查用户是否可以订阅指定主题
     */
    public boolean canSubscribe(String userId, String topic) {
        return checkPermission(userId, topic, "sub");
    }
    
    private boolean checkPermission(String userId, String topic, String action) {
        // 1. 检查Redis缓存
        String cacheKey = "mqtt:acl:" + userId + ":" + action + ":" + topic;
        String cached = redisTemplate.opsForValue().get(cacheKey);
        if (cached != null) {
            return "allow".equals(cached);
        }
        
        // 2. 遍历规则匹配
        for (AclRule rule : rules.values()) {
            if (rule.matches(userId, topic, action)) {
                String result = rule.isAllow() ? "allow" : "deny";
                // 缓存结果 1小时
                redisTemplate.opsForValue().set(cacheKey, result, 1, TimeUnit.HOURS);
                return rule.isAllow();
            }
        }
        
        // 3. 默认策略
        return isDefaultAllow();
    }
    
    private void loadRulesFromFile() {
        try {
            // 加载ACL规则文件 (类似mosquitto acl_file格式)

            // 实现文件加载逻辑
        } catch (Exception e) {
            logger.error("Failed to load ACL rules", e);
        }
    }
    
    private boolean isDefaultAllow() {
        return false; // 默认拒绝
    }
}
```

#### Enhancement #2: Prometheus监控指标

```java
@Component
public class MqttMetricsRegistry {
    
    private final MeterRegistry meterRegistry;
    
    private Counter publishCounter;
    private Counter subscribeCounter;
    private AtomicInteger activeConnections;
    private Timer messageLatency;
    
    @PostConstruct
    public void init() {
        publishCounter = Counter.builder("mqtt.publish.total")
            .description("Total MQTT publish messages")
            .register(meterRegistry);
            
        activeConnections = meterRegistry.gauge("mqtt.connections.active", 
            new AtomicInteger(0));
            
        messageLatency = Timer.builder("mqtt.message.latency")
            .description("MQTT message processing latency")
            .publishPercentiles(0.5, 0.95, 0.99)
            .register(meterRegistry);
    }
    
    public void recordPublish() {
        publishCounter.increment();
    }
    
    public void recordMessageLatency(long duration) {
        messageLatency.record(duration, TimeUnit.MILLISECONDS);
    }
}
```
